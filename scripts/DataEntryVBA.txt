Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    Dim sheetName As String
    sheetName = Sh.Name
    
    If sheetName <> "Personal Observations" And _
       sheetName <> "Commercial Observations" And _
       sheetName <> "Insect Collection" Then Exit Sub
    
    If Target.Column <> 1 Then Exit Sub
    If Target.Row < 2 Then Exit Sub
    If Target.Cells.Count > 1 Then Exit Sub
    
    Dim species As String
    species = Trim(Target.Value)
    
    Dim colTVK As Long, colCommon As Long, colClass As Long
    Dim colOrder As Long, colFamily As Long, colSort As Long
    Dim hasClass As Boolean
    
    Select Case sheetName
        Case "Personal Observations"
            colTVK = 20: colCommon = 21: colClass = 22
            colOrder = 23: colFamily = 24: colSort = 25
            hasClass = True
        Case "Commercial Observations"
            colTVK = 23: colCommon = 24: colClass = 25
            colOrder = 26: colFamily = 27: colSort = 28
            hasClass = True
        Case "Insect Collection"
            colTVK = 21: colCommon = 22: colClass = 0
            colOrder = 23: colFamily = 24: colSort = 25
            hasClass = False
    End Select
    
    If Len(species) = 0 Then
        Sh.Cells(Target.Row, 2).Value = ""
        Sh.Cells(Target.Row, 3).Value = ""
        Sh.Cells(Target.Row, 4).Value = ""
        Sh.Cells(Target.Row, 5).Value = ""
        Sh.Cells(Target.Row, 6).Value = ""
        Sh.Cells(Target.Row, 7).Value = ""
        Sh.Cells(Target.Row, colTVK).Value = ""
        Sh.Cells(Target.Row, colCommon).Value = ""
        If hasClass Then Sh.Cells(Target.Row, colClass).Value = ""
        Sh.Cells(Target.Row, colOrder).Value = ""
        Sh.Cells(Target.Row, colFamily).Value = ""
        Sh.Cells(Target.Row, colSort).Value = ""
        Exit Sub
    End If
    
    Dim uksiRange As Range
    Set uksiRange = Worksheets("UKSI").Range("A:N")
    
    Dim tvk As Variant
    tvk = Application.VLookup(species, uksiRange, 2, False)
    
    If Not IsError(tvk) Then
        Application.EnableEvents = False
        Sh.Cells(Target.Row, 2).Value = Application.VLookup(species, uksiRange, 7, False)
        Sh.Cells(Target.Row, 3).Value = Application.VLookup(species, uksiRange, 8, False)
        Sh.Cells(Target.Row, 4).Value = Application.VLookup(species, uksiRange, 9, False)
        Sh.Cells(Target.Row, 5).Value = Application.VLookup(species, uksiRange, 10, False)
        Sh.Cells(Target.Row, 6).Value = Application.VLookup(species, uksiRange, 11, False)
        Sh.Cells(Target.Row, 7).Value = Application.VLookup(species, uksiRange, 12, False)
        Sh.Cells(Target.Row, colTVK).Value = tvk
        Sh.Cells(Target.Row, colCommon).Value = Application.VLookup(species, uksiRange, 3, False)
        If hasClass Then
            Sh.Cells(Target.Row, colClass).Value = Application.VLookup(species, uksiRange, 4, False)
        End If
        Sh.Cells(Target.Row, colOrder).Value = Application.VLookup(species, uksiRange, 6, False)
        Sh.Cells(Target.Row, colFamily).Value = Application.VLookup(species, uksiRange, 5, False)
        Sh.Cells(Target.Row, colSort).Value = Application.VLookup(species, uksiRange, 13, False)
        Application.EnableEvents = True
        Exit Sub
    End If
    
    If Len(species) >= 2 Then
        ShowSpeciesSearch species, Target.Row, sheetName
    End If
End Sub

Public Sub ShowSpeciesSearch(searchTerm As String, targetRow As Long, sheetName As String)
    Dim ws As Worksheet
    Set ws = Worksheets("UKSI")
    
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    Dim terms() As String
    terms = Split(LCase(Trim(searchTerm)), " ")
    
    Dim matches() As String
    Dim matchRows() As Long
    Dim matchDetails() As String
    Dim matchCount As Long
    matchCount = 0
    ReDim matches(1 To 50)
    ReDim matchRows(1 To 50)
    ReDim matchDetails(1 To 50)
    
    Dim i As Long
    For i = 2 To lastRow
        If matchCount >= 50 Then Exit For
        
        Dim displayName As String
        Dim commonName As String
        displayName = LCase(Trim(ws.Cells(i, 1).Value))
        commonName = LCase(Trim(ws.Cells(i, 3).Value))
        
        Dim allMatch As Boolean
        allMatch = True
        
        Dim t As Long
        For t = 0 To UBound(terms)
            If Len(terms(t)) > 0 Then
                If InStr(1, displayName, terms(t), vbTextCompare) = 0 And _
                   InStr(1, commonName, terms(t), vbTextCompare) = 0 Then
                    allMatch = False
                    Exit For
                End If
            End If
        Next t
        
        If allMatch Then
            matchCount = matchCount + 1
            matches(matchCount) = ws.Cells(i, 1).Value
            matchRows(matchCount) = i
            
            Dim detail As String
            detail = ws.Cells(i, 1).Value
            
            If Len(ws.Cells(i, 3).Value) > 0 Then
                detail = detail & "  (" & ws.Cells(i, 3).Value & ")"
            End If
            
            Dim famOrd As String
            famOrd = ""
            If Len(ws.Cells(i, 5).Value) > 0 Then famOrd = ws.Cells(i, 5).Value
            If Len(ws.Cells(i, 6).Value) > 0 Then
                If Len(famOrd) > 0 Then
                    famOrd = famOrd & ", " & ws.Cells(i, 6).Value
                Else
                    famOrd = ws.Cells(i, 6).Value
                End If
            End If
            If Len(famOrd) > 0 Then detail = detail & "  [" & famOrd & "]"
            
            Dim status As String
            status = ""
            If Len(ws.Cells(i, 8).Value) > 0 Then status = ws.Cells(i, 8).Value
            If Len(ws.Cells(i, 9).Value) > 0 Then
                If Len(status) > 0 Then status = status & ", " & ws.Cells(i, 9).Value Else status = ws.Cells(i, 9).Value
            End If
            If Len(ws.Cells(i, 10).Value) > 0 Then
                If Len(status) > 0 Then status = status & ", S41" Else status = "S41"
            End If
            If Len(ws.Cells(i, 11).Value) > 0 Then
                If Len(status) > 0 Then status = status & ", Legal" Else status = "Legal"
            End If
            If Len(status) > 0 Then detail = detail & "  *" & status & "*"
            
            matchDetails(matchCount) = detail
        End If
    Next i
    
    If matchCount = 0 Then
        MsgBox "No species found matching '" & searchTerm & "'." & vbCrLf & vbCrLf & _
               "Check the UKSI sheet for the correct name.", _
               vbExclamation, "Species Not Found"
        Exit Sub
    End If
    
    If matchCount = 1 Then
        ApplySpeciesSelection matchRows(1), targetRow, sheetName
        Exit Sub
    End If
    
    Dim msg As String
    Dim maxShow As Long
    maxShow = matchCount
    If maxShow > 20 Then maxShow = 20
    
    msg = matchCount & " matches for '" & searchTerm & "'" & vbCrLf
    msg = msg & "Enter number to select:" & vbCrLf & vbCrLf
    
    For i = 1 To maxShow
        msg = msg & i & ".  " & matchDetails(i) & vbCrLf
    Next i
    
    If matchCount > 20 Then
        msg = msg & vbCrLf & "(" & (matchCount - 20) & " more - type more characters to narrow)"
    End If
    
    Dim selection As String
    selection = InputBox(msg, "Select Species", "1")
    
    If Len(selection) = 0 Then Exit Sub
    
    Dim selNum As Long
    On Error Resume Next
    selNum = CLng(selection)
    On Error GoTo 0
    
    If selNum < 1 Or selNum > maxShow Then
        MsgBox "Invalid selection.", vbExclamation
        Exit Sub
    End If
    
    ApplySpeciesSelection matchRows(selNum), targetRow, sheetName
End Sub


Private Sub ApplySpeciesSelection(uksiRow As Long, targetRow As Long, sheetName As String)
    Dim ws As Worksheet
    Dim dataWs As Worksheet
    Set ws = Worksheets("UKSI")
    Set dataWs = Worksheets(sheetName)
    
    Dim colTVK As Long, colCommon As Long, colClass As Long
    Dim colOrder As Long, colFamily As Long, colSort As Long
    Dim hasClass As Boolean
    
    Select Case sheetName
        Case "Personal Observations"
            colTVK = 20: colCommon = 21: colClass = 22
            colOrder = 23: colFamily = 24: colSort = 25
            hasClass = True
        Case "Commercial Observations"
            colTVK = 23: colCommon = 24: colClass = 25
            colOrder = 26: colFamily = 27: colSort = 28
            hasClass = True
        Case "Insect Collection"
            colTVK = 21: colCommon = 22: colClass = 0
            colOrder = 23: colFamily = 24: colSort = 25
            hasClass = False
    End Select
    
    Application.EnableEvents = False
    
    dataWs.Cells(targetRow, 1).Value = ws.Cells(uksiRow, 1).Value
    dataWs.Cells(targetRow, 2).Value = ws.Cells(uksiRow, 7).Value
    dataWs.Cells(targetRow, 3).Value = ws.Cells(uksiRow, 8).Value
    dataWs.Cells(targetRow, 4).Value = ws.Cells(uksiRow, 9).Value
    dataWs.Cells(targetRow, 5).Value = ws.Cells(uksiRow, 10).Value
    dataWs.Cells(targetRow, 6).Value = ws.Cells(uksiRow, 11).Value
    dataWs.Cells(targetRow, 7).Value = ws.Cells(uksiRow, 12).Value
    
    dataWs.Cells(targetRow, colTVK).Value = ws.Cells(uksiRow, 2).Value
    dataWs.Cells(targetRow, colCommon).Value = ws.Cells(uksiRow, 3).Value
    If hasClass Then
        dataWs.Cells(targetRow, colClass).Value = ws.Cells(uksiRow, 4).Value
    End If
    dataWs.Cells(targetRow, colOrder).Value = ws.Cells(uksiRow, 6).Value
    dataWs.Cells(targetRow, colFamily).Value = ws.Cells(uksiRow, 5).Value
    dataWs.Cells(targetRow, colSort).Value = ws.Cells(uksiRow, 13).Value
    
    Application.EnableEvents = True
End Sub
