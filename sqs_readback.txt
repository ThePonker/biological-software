  428:     def _derive_sqs_batch(self, tvks):
  429:         """Apply Pantheon's published rule to current Codex statuses.
  430: 
  431:         Invertebrates only -- SQS is an invertebrate construct, and a score for
  432:         a lichen would be arithmetic without meaning.
  433: 
  434:         Queries status_summary directly: get_statuses_batch calls
  435:         get_sqs_scores, so routing through it would recurse.
  436:         """
  437:         if not tvks:
  438:             return {}
  439:         try:
  440:             from shared.sqs_derivation import derive_from_tracks
  441:         except ImportError:
  442:             try:
  443:                 from sqs_derivation import derive_from_tracks
  444:             except ImportError:
  445:                 return {}
  446: 
  447:         invert = self._get_invert_tvks()
  448:         candidates = [t for t in tvks if t in invert]
  449:         if not candidates:
  450:             return {}
  451: 
  452:         c = self._get_conn().cursor()
  453:         tracks = {}
  454:         for batch in _chunked(candidates, 500):
  455:             ph = ",".join("?" * len(batch))
  456:             c.execute(f"""SELECT tvk, status_track, status_value
  457:                           FROM status_summary WHERE tvk IN ({ph})""", batch)
  458:             for row in c.fetchall():
  459:                 tracks.setdefault(row["tvk"], {})[row["status_track"]] = row["status_value"]
  460: 
  461:         result = {}
  462:         for tvk in candidates:
  463:             score = derive_from_tracks(tracks.get(tvk, {}))
